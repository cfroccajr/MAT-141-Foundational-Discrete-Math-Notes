\begin{thm}
    Every tree has a vertex of degree 1.
\end{thm}

\begin{thm}
    A tree with \(n\) vertices has \(n-1\) edges.
\end{thm}

\begin{defn}[Binary Trees]
\hspace*{1in}
\begin{itemize}
    \item A \emph{binary tree} is a tree in which each parent has\newline \framebox{\phantom{\huge at most two children.}}
    \item A \emph{full binary tree} is a tree in which each parent has\newline \framebox{\phantom{\huge exactly two children.}}
    % \item A \emph{complete binary tree} is a \framebox{\phantom{\Large full binary tree.}} in which all children\newline \framebox{\phantom{\Large are at the same height.}}

\begin{thm}
    Given a full binary tree with \(k\) internal vertices, there are \(n=2k+1\) vertices all together and so \(k+1\) leaves.
    \begin{center}
        \begin{tikzpicture}
        \graph[ 
            binary tree layout,
            nodes={vertex,color=white},
            edges={edges},
            level distance=1cm,
            sibling distance=2.5cm
        ]
        {
            a[ball color=accent]--{
                b[ball color=secondaryaccent]--{
                    d[ball color=secondaryaccent]--{
                        i[ball color=tertiaryaccent],
                        j[ball color=tertiaryaccent]
                    },
                    e[ball color=tertiaryaccent]
                },
                c[ball color=secondaryaccent]--{
                    f[ball color=secondaryaccent]--{
                        k[ball color=tertiaryaccent],
                        l[ball color=tertiaryaccent]
                    },
                    g[ball color=secondaryaccent]--{
                        m[ball color=tertiaryaccent],
                        n[ball color=tertiaryaccent]
                    }
                }
            }
        };
        \end{tikzpicture}
    \end{center}
\end{thm}

\begin{thm}
    Given a binary tree of height \(h\) with \(t\) leaves, \emph{\(t\leq 2^h\)} or equivalently \emph{\(\log_2(t)\leq h\)}.
    Equality occurs in full binary trees in which all leaves are the same height.
\end{thm}

    \item There exists a tree with 7 vertices and 7 edges.
