\subsection*{Further Notes:}

\noindent
\begin{enumerate}
\setcounter{enumi}{21}
	\item $p$ is sufficient for $q$ $\equiv$ $p\rightarrow q\ \equiv\ p,$ only if $q$.
	\item $p$ is necessary for $q$ $\equiv$ $q\rightarrow p\ \equiv\ p,$ if $q$.
	\item $p$ is necessary and sufficient for $q$ $\equiv$ $p\leftrightarrow q\ \equiv\ p,$ if and only if $q$.
	\item Quantifier Negation (QN): \(\forall\equiv\ for\ all\) and \(\exists\equiv\ there\ exists\)
\begin{itemize}
	\item[(a)] $\sim \left[\left(\forall x\right)\left(p(x)\right)\right]\equiv\left(\exists x\right)\left(\sim p(x)\right)$
	\item[(b)] $\sim \left[\left(\exists x\right)\left(p(x)\right)\right]\equiv\left(\forall x\right)\left(\sim p(x)\right)$
\end{itemize}
	\item Principle of Mathematical Induction (PMI): If $P(n)$ is a statement about or involving the natural number $n$ such that
\begin{itemize}
	\item[(1)] $P(1)$ is true, and
	\item[(2)] $P(k)\Rightarrow P(k+1)$ for all natural numbers $k$, then
\end{itemize}
$P(n)$ is true for all natural numbers $n$.

\noindent
	\item Four basic truth tables:

\begin{multicols}{2}
\begin{enumerate}
	\item[(a)] Negation\\[5pt]
	$\begin{array}{c|c}
		p & \sim p \\ \hline
		T & F \\
		F & T \\
	\end{array}$\vspace{20pt}
	\item[(b)] Conjunction\\[5pt]
	$\begin{array}{c|c|c}
		p & q & p\land q \\ \hline
		T & T & T\\
		T & F & F\\
		F & T & F\\
		F & F & F\\
	\end{array}$
\columnbreak
        \item[(c)] Disjunction\\[5pt]
	$\begin{array}{c|c|c}
		p & q & p\lor q \\ \hline
		T & T & T\\
		T & F & T\\
		F & T & T\\
		F & F & F\\
	\end{array}$
	\item[(d)] Conditional\\[5pt]
	$\begin{array}{c|c|c}
		p & q & p\rightarrow q \\ \hline
		T & T & T\\
		T & F & F\\
		F & T & T\\
		F & F & T\\
	\end{array}$
\end{enumerate}
\end{multicols}

	\item Converse Error \((P\rightarrow Q) \wedge Q \not\Rightarrow P\):

	\begin{array}{c|c|c}

			\rowcolor{gray!50!white} F & T & T\\
			F & F & T\\
	\end{array}
\]

			\rowcolor{gray!50!white} F & F & T\\
	\end{array}
\]
\end{enumerate}
