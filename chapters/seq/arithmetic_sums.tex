    \begin{align*}
        S_{100}&=1+2+3+\cdots+98+99+100\\
        S_{100}&=100+99+98+\cdots+3+2+1\\[15pt]

        2\cdot S_{k}&= \\[15pt]

    \end{align*}

% Arithmetic Sum Definition

\begin{defn}[Arithmetic Sum]\index{arithmetic sum}
    An \emph{arithmetic sum} is a summation of the form
    \[\sum_{i=1}^n ai+b\]
    which has the propery that consecutive terms have a constant difference, \(a\).
\end{defn}

% Arithmetic Sum, General
\begin{exposition}[General Arithmetic Sum]\label{exp:gen_arith}\index{arithmetic sum} Finally consider the general arithmetic summation \[S_n=\sum_{i=1}^n ai+b=(a+b)+(2a+b)+(3a+b)+\cdots+(n\cdot a+b).\]

For \(n=k\) we can again use the same strategy as Exposition \ref{exp:arithmetic} to find a formula for this sum.
    \begin{align*}
        S_{k}&=(a+b)+(2a+b)+(3a+b)+\cdots+(k\cdot a+b)\\
        S_{k}&=(k\cdot a+b)+\cdots+(3a+b)+(2a+b)+(a+b)\\[30pt]
        2\cdot S_{k}&= \\[30pt]
        S_{k}&=\\[30pt]
        S_{k}&=
    \end{align*}
    \vspace{\baselineskip}
\end{exposition}
