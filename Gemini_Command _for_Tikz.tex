#!/bin/bash

# Exit on severe errors
set -e

OUTPUT_DIR="../generated_images"
WRAPPER_FILE="../wrapper.tex"

# Ensure output directory exists
mkdir -p "$OUTPUT_DIR"

# Verify wrapper file exists
if [ ! -f "$WRAPPER_FILE" ]; then
  echo "Error: Wrapper file '$WRAPPER_FILE' not found!"
  exit 1
fi

echo "Starting compilation and conversion..."

# Process each .tex file
for f in *.tex; do
  # Skip loop if no .tex files are found
  [ -e "$f" ] || continue

  BASENAME="${f%.tex}"
  echo "Processing: $f -> ${BASENAME}.pdf & ${BASENAME}.png"

  # 1. Compile TeX to PDF
  lualatex -interaction=nonstopmode --shell-escape \
    --output-directory="$OUTPUT_DIR" \
    --jobname="$BASENAME" \
    "\def\TargetFile{$f}\input{$WRAPPER_FILE}" > /dev/null

  # 2. Convert PDF to PNG (300 DPI)
  magick -density 300 "$OUTPUT_DIR/${BASENAME}.pdf" "$OUTPUT_DIR/${BASENAME}.png"

  # 3. Clean up intermediate TeX auxiliary files
  rm -f "$OUTPUT_DIR/${BASENAME}".{aux,log,dpth,md5}
done

echo "Done! All images saved to $OUTPUT_DIR"
