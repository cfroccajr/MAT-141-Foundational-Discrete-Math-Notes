\begin{tikzpicture}[scale=1,transform shape]
    \foreach \n in {0,...,5}{
        \node at (0,-\n) {\(n=\n\)};
        \pgfmathparse{int(factorial(\n))}\let\numer=\pgfmathresult;
        \foreach \k in {0,...,\n}{
            \pgfmathparse{div(\numer,(factorial(\k)*factorial(\n-\k))}\let\val=\pgfmathresult;
            \node at (\k-\n/2+5,-\n) {\val};
        }
    }
    \draw[->,shorten <=10pt,shorten >=10pt] (5,-4) -- (4.5,-5);
    \draw[->,shorten <=10pt,shorten >=10pt] (4,-4) -- (4.5,-5);
\end{tikzpicture}
